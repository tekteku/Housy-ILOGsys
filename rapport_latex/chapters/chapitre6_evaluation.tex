\section{Métriques d'Évaluation}

\subsection{Métriques de Performance Technique}

\subsubsection{Métriques de Régression}

\begin{itemize}
    \item \textbf{R² (Coefficient de Détermination)} : Mesure la proportion de variance expliquée
    \item \textbf{RMSE (Root Mean Square Error)} : Erreur quadratique moyenne
    \item \textbf{MAE (Mean Absolute Error)} : Erreur absolue moyenne
    \item \textbf{MAPE (Mean Absolute Percentage Error)} : Erreur absolue en pourcentage
    \item \textbf{RMSLE (Root Mean Square Logarithmic Error)} : RMSE sur échelle logarithmique
\end{itemize}

\subsubsection{Formules Mathématiques}

\begin{align}
R^2 &= 1 - \frac{\sum_{i=1}^{n}(y_i - \hat{y}_i)^2}{\sum_{i=1}^{n}(y_i - \bar{y})^2} \\
RMSE &= \sqrt{\frac{1}{n}\sum_{i=1}^{n}(y_i - \hat{y}_i)^2} \\
MAE &= \frac{1}{n}\sum_{i=1}^{n}|y_i - \hat{y}_i| \\
MAPE &= \frac{100\%}{n}\sum_{i=1}^{n}\left|\frac{y_i - \hat{y}_i}{y_i}\right|
\end{align}

\section{Tests de Robustesse}

\subsection{Test de Sensibilité aux Outliers}

\begin{lstlisting}[language=Python, caption=Test de robustesse aux outliers]
import numpy as np
from sklearn.metrics import mean_absolute_percentage_error

def test_outlier_sensitivity(model, X_test, y_test, contamination_rate=0.05):
    """Test la sensibilité du modèle aux valeurs aberrantes"""
    
    # Prédictions sur données originales
    original_pred = model.predict(X_test)
    original_mape = mean_absolute_percentage_error(y_test, original_pred)
    
    # Injection d'outliers
    n_outliers = int(len(X_test) * contamination_rate)
    outlier_indices = np.random.choice(len(X_test), n_outliers, replace=False)
    
    X_test_contaminated = X_test.copy()
    # Multiplication par facteur aléatoire pour créer des outliers
    for idx in outlier_indices:
        factor = np.random.uniform(2, 5)
        X_test_contaminated.iloc[idx] *= factor
    
    # Prédictions sur données contaminées
    contaminated_pred = model.predict(X_test_contaminated)
    contaminated_mape = mean_absolute_percentage_error(y_test, contaminated_pred)
    
    # Calcul de la dégradation
    degradation = (contaminated_mape - original_mape) / original_mape * 100
    
    return {
        'original_mape': original_mape,
        'contaminated_mape': contaminated_mape,
        'degradation_percent': degradation
    }

# Résultats du test
robustness_results = test_outlier_sensitivity(ensemble_model, X_test, y_test)
print(f"MAPE original: {robustness_results['original_mape']:.1%}")
print(f"MAPE avec outliers: {robustness_results['contaminated_mape']:.1%}")
print(f"Dégradation: {robustness_results['degradation_percent']:.1f}%")
\end{lstlisting}

\subsection{Monitoring de la Dérive du Modèle}

\begin{lstlisting}[language=Python, caption=Monitoring de la dérive]
import numpy as np
from scipy import stats

def detect_data_drift(reference_data, current_data, threshold=0.05):
    """Détection de dérive des données"""
    
    drift_detected = {}
    
    for column in reference_data.columns:
        if reference_data[column].dtype in ['float64', 'int64']:
            # Test de Kolmogorov-Smirnov pour variables continues
            statistic, p_value = stats.ks_2samp(
                reference_data[column], 
                current_data[column]
            )
        else:
            # Test du Chi² pour variables catégorielles
            ref_counts = reference_data[column].value_counts()
            curr_counts = current_data[column].value_counts()
            
            # Alignement des comptes
            all_categories = set(ref_counts.index) | set(curr_counts.index)
            ref_aligned = [ref_counts.get(cat, 0) for cat in all_categories]
            curr_aligned = [curr_counts.get(cat, 0) for cat in all_categories]
            
            statistic, p_value = stats.chisquare(curr_aligned, ref_aligned)
        
        drift_detected[column] = {
            'statistic': statistic,
            'p_value': p_value,
            'drift_detected': p_value < threshold
        }
    
    return drift_detected

# Surveillance mensuelle
monthly_drift_results = detect_data_drift(training_data, production_data)
\end{lstlisting}
